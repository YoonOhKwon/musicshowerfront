function average(
  array
) {

  if (
    !array ||
    !array.length
  ) {

    return 0;

  }


  return (
    array.reduce(
      (a, b) =>
        a + b,
      0
    ) /
    array.length
  );

}

    if (
      !Array.isArray(
        matrix[name]
      )
    ) {

      throw new Error(
        `Missing category: ${name}`
      );

    }

  }


  if (
    !matrix.genreProfile
  ) {

    matrix.genreProfile = {

      family:
        "unknown",

      primary:
        "unknown",

      secondary:
        [],

      confidence:
        0

  if (
    typeof matrix.intensity !==
    "number"

    matrix.intensity =
      0.5;

  }


  matrix.intensity =
    constrain(
      matrix.intensity,
      0,
      1
    );

}

/* =========================
   FREQUENCY BANDS
========================= */

function calculateFrequencyBands() {

  if (
    !audioFrequencyData ||
    !audioContext ||
    !analyser
  ) {

    return;

  }


  const length =
    audioFrequencyData.length;


  let subBass = 0;

  let bass = 0;

  let lowMid = 0;

  let mid = 0;

  let high = 0;


  for (
    let i = 0;
    i < length;
    i++
  ) {

    const value =
      audioFrequencyData[i] /
      255;


    const frequency =
      i *
      (
        audioContext.sampleRate /
        analyser.fftSize
      );


    if (
      frequency < 60
    ) {

      subBass += value;

    }

    else if (
      frequency < 250
    ) {

      bass += value;

    }

    else if (
      frequency < 500
    ) {

      lowMid += value;

    }

    else if (
      frequency < 2000
    ) {

      mid += value;

    }

    else {

      high += value;

    }

  }


  pushLimited(
    audioAnalysis.bassList,
    (
      subBass +
      bass
    ) / length
  );


  pushLimited(
    audioAnalysis.lowMidList,
    lowMid / length
  );


  pushLimited(
    audioAnalysis.midList,
    mid / length
  );


  pushLimited(
    audioAnalysis.highList,
    high / length
  );

}


/* =========================
   STATISTICS
========================= */

function getFeatureStats(
  array
) {

  if (
    !array ||
    !array.length
  ) {

    return {

      mean: 0,

      std: 0,

      min: 0,

      max: 0,

      range: 0

    };

  }


  const mean =
    array.reduce(
      (a, b) => a + b,
      0
    ) /
    array.length;


  const variance =
    array.reduce(
      (sum, value) =>
        sum +
        Math.pow(
          value - mean,
          2
        ),
      0
    ) /
    array.length;


  const min =
    Math.min(
      ...array
    );


  const max =
    Math.max(
      ...array
    );


  return {

    mean,

    std:
      Math.sqrt(
        variance
      ),

    min,

    max,

    range:
      max - min

  };

}

function getChromaProfile() {

  ) {

      let i = 0;

    ) {

      result[i] +=
        Number(
          frame[i]
        ) || 0;

    }

  }


  return result.map(
    value =>
      value /
      history.length
  );

}

/* =========================

function averageFrequency(
  start,
  end
) {

  if (
    !audioFrequencyData
  ) {

    return 0;

  }


  end =
    Math.min(
      end,
      audioFrequencyData.length
    );


  let sum = 0;


  for (
    let i = start;
    i < end;
    i++
  ) {

    sum +=
      audioFrequencyData[i];

  }


  return (
    sum /
    Math.max(
      1,
      end - start
    )
  );

}


/* =========================
   BEAT
========================= */

function updateBeatDetection() {

  if (
    !audioFrequencyData
  ) {

    return false;

  }


  const bass =
    averageFrequency(0, 8);


  const mid =
    averageFrequency(8, 35);


  const high =
    averageFrequency(35, 100);


  const energy =
    (
      bass * 0.5 +
      mid * 0.3 +
      high * 0.2
    ) / 255;


  smoothBass =
    lerp(
      smoothBass,
      bass,
      0.18
    );


  smoothMid =
    lerp(
      smoothMid,
      mid,
      0.14
    );


  smoothHigh =
    lerp(
      smoothHigh,
      high,
      0.14
    );


  smoothEnergy =
    lerp(
      smoothEnergy,
      energy,
      0.12
    );


  const bassDelta =
    smoothBass -
    previousBass;


  const energyDelta =
    smoothEnergy -
    previousEnergy;


  const now =
    performance.now();


  const threshold =
    CONFIG.beat.bassDelta +
    smoothBass * 0.12;


  let detected =
    false;


  if (

    bassDelta >
    threshold &&

    energyDelta >
    CONFIG.beat.energyDelta &&

    smoothBass >
    CONFIG.beat.minBass &&

    now - lastBeatTime >
    CONFIG.beat.cooldown

  ) {

    detected =
      true;


    if (
      lastBeatTime > 0
    ) {

      const interval =
        now -
        lastBeatTime;


      updateBPMFromInterval(
        interval
      );

    }


    lastBeatTime =
      now;


    beatFlash =
      1;

    beatScale =
      1;

  }


  previousBass =
    smoothBass;


  previousEnergy =
    smoothEnergy;


  beatFlash *=
    0.92;


  beatScale *=
    0.91;


  return detected;

}


/* =========================
   BPM
========================= */

function updateBPMFromInterval(
  interval
) {

  if (
    interval < 250 ||
    interval > 2000
  ) {

    return;

  }


  beatIntervalHistory.push(
    interval
  );


  if (
    beatIntervalHistory.length >
    16
  ) {

    beatIntervalHistory.shift();

  }


  const estimated =
    60000 / interval;


  if (
    estimated <
    CONFIG.beat.bpmMin ||

    estimated >
    CONFIG.beat.bpmMax
  ) {

    return;

  }


  bpmHistory.push(
    estimated
  );


  if (
    bpmHistory.length >
    8
  ) {

    bpmHistory.shift();

  }


  bpm =
    bpmHistory.reduce(
      (a, b) =>
        a + b,
      0
    ) /
    bpmHistory.length;

}


/* =========================
   RHYTHM PROFILE
========================= */

function getRhythmProfile() {

  if (
    !beatIntervalHistory.length
  ) {

    return {

      bpm:
        bpm || 0,

      bpmStability:
        0,

      beatIntervalMean:
        0,

      beatIntervalStd:
        0,

      beatCount:
        0

    };

  }


  const mean =
    beatIntervalHistory.reduce(
      (a, b) =>
        a + b,
      0
    ) /
    beatIntervalHistory.length;


  const variance =
    beatIntervalHistory.reduce(
      (sum, value) =>
        sum +
        Math.pow(
          value - mean,
          2
        ),
      0
    ) /
    beatIntervalHistory.length;


  const std =
    Math.sqrt(
      variance
    );


  const coefficient =
    mean > 0
      ? std / mean
      : 1;


  const stability =
    constrain(
      1 - coefficient,
      0,
      1
    );


  return {

    bpm:
      bpm || 0,

    bpmStability:
      stability,

    beatIntervalMean:
      mean,

    beatIntervalStd:
      std,

    beatCount:
      beatIntervalHistory.length

  };

}
